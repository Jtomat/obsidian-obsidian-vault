```python
from __future__ import annotations  
  
from enum import Enum  
from typing import Optional, Dict  
  
import torch  
from pydantic import BaseModel  
  
  
# ------------[Абстрактное ядро]--------------  
  
class Node(BaseModel):  
    type: str  
  
  
class Context:  
    declarations: Dict[str, Declaration]  
  
    def get_declaration(self, name: str) -> int | float | torch.tensor:  
        return self.declarations[name].value.eval(self)  
  
    def set_declaration(self, declaration: Declaration) -> None:  
        self.declarations[declaration.name] = declaration  
  
  
# Вычисляемый элемент  
class Expression(Node):  
  
    def eval(self, context: Context) -> int | float | torch.tensor:  
        pass  
  
  
# Запись значения в контекст по имени  
class Declaration(Node):  
    name: str  
    value: Expression  
  
    def eval(self, context: Context) -> None:  
        context.set_declaration(self)  
  
```

```python
    type: str = 'reference'

    def eval(self, context: Context, local: Optional[Context] = None) -> int | float | torch.tensor:
```

```python
# --------------[Унарная матеша]--------------  
  
class UnaryOperation(Enum):  
    Negate = "negate"  
    Abs = "abs"  
    Sqrt = "sqrt"  
    Sin = "sin"  
    Cos = "cos"  
    Tan = "tan"  
    Exp = "exp"  
    Log = "log"  
    Floor = "floor"  
    Ceil = "ceil"  
  
  
import math  
  
UNARY_OPERATORS_FUNCS = {  
    UnaryOperation.Negate: (torch.neg, lambda x: -x),  
    UnaryOperation.Abs: (torch.abs, abs),  
    UnaryOperation.Sqrt: (torch.sqrt, math.sqrt),  
    UnaryOperation.Sin: (torch.sin, math.sin),  
    UnaryOperation.Cos: (torch.cos, math.cos),  
    UnaryOperation.Tan: (torch.tan, math.tan),  
    UnaryOperation.Exp: (torch.exp, math.exp),  
    UnaryOperation.Log: (torch.log, math.log),  
    UnaryOperation.Floor: (torch.floor, math.floor),  
    UnaryOperation.Ceil: (torch.ceil, math.ceil),  
}  
  
  
class UnaryMathOperationNode(Expression):  
    type: str = 'unary_math_operation'  
  
    operator: UnaryOperation  
    operand: Expression  
  
    def eval(self, context: Context, local: Optional[Context] = None) -> int | float | torch.tensor:  
        base_val = self.operand.eval(context)  
        to_exec = UNARY_OPERATORS_FUNCS[self.operator][int(not torch.is_tensor(base_val))]  
  
        return to_exec(base_val)  
  
```

```python  
# --------------[Бинарная матеша]--------------  
  
class BinaryOperation(Enum):  
    Add = "add"  
    Sub = "sub"  
    Mul = "mul"  
    Div = "div"  
    Mod = "mod"  
    Pow = "pow"  
    Min = "min"  
    Max = "max"  
  
  
import operator  
  
BINARY_OPERATORS_FUNCS = {  
    BinaryOperation.Add: (torch.add, operator.add),  
    BinaryOperation.Sub: (torch.sub, operator.sub),  
    BinaryOperation.Mul: (torch.mul, operator.mul),  
    BinaryOperation.Div: (torch.div, operator.truediv),  
    BinaryOperation.Mod: (torch.remainder, operator.mod),  
    BinaryOperation.Pow: (torch.pow, operator.pow),  
    BinaryOperation.Min: (torch.minimum, min),  
    BinaryOperation.Max: (torch.maximum, max),  
}  
  
  
class BinaryOperationNode(Expression):  
    type: str = "binary_math_operation"  
  
    operator: BinaryOperation  
  
    left: Expression  
    right: Expression  
  
    def eval(self, context: Context, local: Optional[Context] = None) -> int | float | torch.tensor:  
        left = self.left.eval(context)  
        right = self.left.eval(context)  
  
        has_tensor = int(not (torch.is_tensor(left) or torch.is_tensor(right)))  
  
        return BINARY_OPERATORS_FUNCS[self.operator][has_tensor](left, right)  
```

```python
class ComparisonOperator(Enum):  
    Equal = "equal"  
    Not_Equal = "not_equal"  
    Less = "less"  
    Less_Equal = "less_equal"  
    Greater = "greater"  
    Greater_Equal = "greater_equal"  
  
import operator  
  
COMPARISON_OPERATORS_FUNCS = {  
    ComparisonOperator.Equal: operator.eq,  
    ComparisonOperator.Not_Equal: operator.ne,  
    ComparisonOperator.Less: operator.lt,  
    ComparisonOperator.Less_Equal: operator.le,  
    ComparisonOperator.Greater: operator.gt,  
    ComparisonOperator.Greater_Equal: operator.ge,  
}  
  
  
class ComparisonNode(Expression):  
    type: str = "comparison"  
  
    operator: ComparisonOperator  
  
    left: Expression  
    right: Expression  
  
    def eval(self, context: Context, local: Optional[Context] = None) -> int | float | torch.tensor:  
        left = self.left.eval(context)  
        right = self.right.eval(context)  
  
        return COMPARISON_OPERATORS_FUNCS[self.operator](left, right)
```

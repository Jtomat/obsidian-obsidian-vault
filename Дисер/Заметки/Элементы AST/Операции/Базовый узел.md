from __future__ import annotations

from typing import Annotated, Literal, Union
from pydantic import BaseModel, Field


class Node(BaseModel):
    type: str


Expression = Annotated[
    Union[
        LiteralNode,
        ReferenceNode,
        UnaryOperationNode,
        BinaryOperationNode,
        FunctionCallNode,
        ConditionalNode,
        ReductionNode,
        IndexNode,
    ],
    Field(discriminator="type"),
]
